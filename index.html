-- Allow new users to request either visitor or admin access. drop policy if exists "Users can submit their own request" on public.access_requests; create policy "Users can submit their own request" on public.access_requests for insert to authenticated with check ( user_id = (select auth.uid()) and status = 'pending' and requested_role in ('visitor', 'admin') ); -- Allow the owner to review either type of request. create or replace function public.review_access_request( target_user_id uuid, allow_access boolean ) returns void language plpgsql security definer set search_path = '' as $$ declare requested_role_value text; begin if not public.is_owner() then raise exception 'Only the owner can review access requests'; end if; select r.requested_role into requested_role_value from public.access_requests r where r.user_id = target_user_id and r.status = 'pending'; if requested_role_value is null then raise exception 'Pending request not found'; end if; update public.access_requests set status = case when allow_access then 'approved' else 'denied' end where user_id = target_user_id; if allow_access then perform public.set_whitelist_access( target_user_id, requested_role_value, true ); else update public.access_whitelist set approved = false where user_id = target_user_id; end if; end; $$; revoke all on function public.review_access_request(uuid, boolean) from public; grant execute on function public.review_access_request(uuid, boolean) to authenticated; -- Check the owner password server-side, not in the HTML. create or replace function public.check_owner_panel_password( candidate text ) returns boolean language sql stable security definer set search_path = '' as $$ select public.is_owner() and candidate = 'destroy'; $$; revoke all on function public.check_owner_panel_password(text) from public; grant execute on function public.check_owner_panel_password(text) to authenticated;
